% !TeX root = ../main.tex
% ============================================================
%  Macro vẽ bàn cờ Tactical Formation (mô phỏng renderer SVG của
%  docs/visual_model.html). \input từ preamble.tex.
%
%  Dùng:
%    \begin{tfboard}            % scale mặc định 1.2
%    \begin{tfboard}[0.9]       % chỉ định scale
%      \Gv{x}{y}  quân đỉnh Xanh     \Gc{x}{y} quân tâm Xanh
%      \Yv{x}{y}  quân đỉnh Vàng     \Yc{x}{y} quân tâm Vàng
%      \hlV{x}{y} / \hlC{x}{y}       vòng đỏ nét đứt (đỉnh / tâm)
%      \jarrow{x1}{y1}{x2}{y2}       mũi tên cú nhảy (đỏ, nét đứt)
%      \ray{x1}{y1}{x2}{y2}          tia ăn (đỏ, mờ)
%      \turn{LƯỢT: Xanh đi}          nhãn lượt dưới bàn
%    \end{tfboard}
%  Toạ độ: đỉnh ở số nguyên 0..4, tâm ở nửa nguyên (i+.5, j+.5).
% ============================================================

\tikzset{
  tfgrid/.style = {gray!45, line width=0.4pt},
  tfvdot/.style = {fill=black!35, draw=none},
  tfcdot/.style = {fill=black!18, draw=none},
}

\newcommand{\tfVR}{0.30} % bán kính quân đỉnh (đơn vị toạ độ)
\newcommand{\tfCR}{0.185}% bán kính quân tâm

% Bàn trống: lưới + chấm mờ tại mọi ô đỉnh/tâm (quân vẽ đè lên trên).
\newcommand{\tfemptyboard}{%
  \foreach \i in {0,...,4}{%
    \draw[tfgrid] (0,\i)--(4,\i);
    \draw[tfgrid] (\i,0)--(\i,4);%
  }%
  \foreach \x in {0,...,4}{\foreach \y in {0,...,4}{%
    \fill[tfvdot] (\x,\y) circle (0.05);}}%
  \foreach \x in {0,...,3}{\foreach \y in {0,...,3}{%
    \fill[tfcdot] (\x+0.5,\y+0.5) circle (0.035);}}%
}

% Quân (Xanh = greenpiece, Vàng = yellowpiece; tâm nhạt + nhỏ hơn đỉnh)
\newcommand{\Gv}[2]{\filldraw[fill=greenpiece, draw=greenpiece!45!black, line width=0.5pt] (#1,#2) circle (\tfVR);}
\newcommand{\Gc}[2]{\filldraw[fill=greenpiece!55,draw=greenpiece!40!black, line width=0.4pt] (#1,#2) circle (\tfCR);}
\newcommand{\Yv}[2]{\filldraw[fill=yellowpiece, draw=yellowpiece!55!black,line width=0.5pt] (#1,#2) circle (\tfVR);}
\newcommand{\Yc}[2]{\filldraw[fill=yellowpiece!65,draw=yellowpiece!55!black,line width=0.4pt] (#1,#2) circle (\tfCR);}

% Vòng nhấn nét đứt
\newcommand{\hlV}[2]{\draw[red!75, line width=0.9pt, dash pattern=on 3pt off 2pt] (#1,#2) circle (0.42);}
\newcommand{\hlC}[2]{\draw[red!75, line width=0.9pt, dash pattern=on 3pt off 2pt] (#1,#2) circle (0.30);}

% Mũi tên cú nhảy + tia ăn
\newcommand{\jarrow}[4]{\draw[-{Stealth}, red!70, line width=1pt, dash pattern=on 4pt off 2pt] (#1,#2) to[bend left=18] (#3,#4);}
\newcommand{\ray}[4]{\draw[red!70, line width=1.8pt, opacity=0.40, line cap=round] (#1,#2)--(#3,#4);}

% Nhãn lượt dưới đáy bàn
\newcommand{\turn}[1]{\node[font=\footnotesize\bfseries, color=vertexcolor!60!black] at (2,-0.55) {#1};}

% Môi trường bàn cờ (đối số tuỳ chọn = scale)
\newenvironment{tfboard}[1][1.2]
  {\begin{tikzpicture}[scale=#1]\tfemptyboard}
  {\end{tikzpicture}}
